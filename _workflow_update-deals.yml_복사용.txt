name: Update hotdeal data

on:
  workflow_dispatch:
  schedule:
    # 30분 단위 자동 갱신: 매시간 7분, 37분 실행
    - cron: "7,37 * * * *"

permissions:
  contents: write

concurrency:
  group: hotdeal-radar-crawl
  cancel-in-progress: true

jobs:
  crawl:
    runs-on: ubuntu-latest
    env:
      PYTHONUNBUFFERED: "1"
      HOTDEAL_RADAR_MAX_PER_SOURCE: "35"
      HOTDEAL_RADAR_DETAIL_LIMIT: "25"
      HOTDEAL_RADAR_DELAY: "1.1"

    steps:
      - name: Checkout
        uses: actions/checkout@v5

      - name: Set up Python
        uses: actions/setup-python@v6
        with:
          python-version: "3.12"
          cache: "pip"

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          python -m pip install -r requirements.txt

      - name: Run crawler
        run: python tools/crawler.py

      - name: Commit updated data
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add data/deals.json data/sources.json
          if git diff --cached --quiet; then
            echo "No changes to commit"
          else
            git commit -m "Update hotdeal data"
            git push
          fi
